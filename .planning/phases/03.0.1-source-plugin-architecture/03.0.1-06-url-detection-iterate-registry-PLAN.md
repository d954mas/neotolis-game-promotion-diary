After refactor, services/url-parser.ts becomes a thin wrapper:
```typescript
// services/url-parser.ts (REFACTORED — orchestrator)
import { parseAnyUrl } from "$lib/sources/url.js";
import { detectFutureKind } from "$lib/sources/future-kinds.js";

export type ParsedUrl =
  | { kind: "youtube_video"; videoId: string; canonicalUrl: string }
  | { kind: "twitter_post"; canonicalUrl: string }
  | { kind: "telegram_post"; canonicalUrl: string }
  | { kind: "reddit_deferred" }
  | { kind: "unsupported" };

const X_HOSTS = new Set(["twitter.com", "x.com", "mobile.twitter.com"]);
const TG_HOSTS = new Set(["t.me"]);

export function parseIngestUrl(input: string): ParsedUrl {
  // 1) Try the adapter registry first (D-15 first-match-wins; YouTube wins).
  const routed = parseAnyUrl(input);
  if (routed.kind === "youtube_video") {
    const meta = routed.metadata ?? {};
    return {
      kind: "youtube_video",
      videoId: routed.externalId,
      canonicalUrl: typeof meta.canonicalUrl === "string" ? meta.canonicalUrl : input,
    };
  }
  // 2) Fallback to host classification for kinds without an adapter yet.
  let url: URL;
  try { url = new URL(input.trim()); } catch { return { kind: "unsupported" }; }
  const host = url.hostname.toLowerCase();
  if (X_HOSTS.has(host))   return { kind: "twitter_post", canonicalUrl: canonicalizeTwitter(input) };
  if (TG_HOSTS.has(host))  return { kind: "telegram_post", canonicalUrl: input };
  // 3) Reddit deferral preservation (RESEARCH.md SOTA divergence #3).
  const future = detectFutureKind(input);
  if (future === "reddit_post") return { kind: "reddit_deferred" };
  return { kind: "unsupported" };
}
```

This preserves the public ParsedUrl union shape (so services/ingest.ts continues to consume it without behavioral change) while routing YouTube through the adapter registry. Twitter, Telegram, Reddit-deferred remain handled at the cross-source url-parser.ts layer until their adapters land.

       // sources/youtube/server/url.ts (EXTENDED — keep existing parseYoutubeUrl + add youtubeParseUrl)
       import type { ParsedUrl } from "$lib/sources/adapter.js";

       const YT_HOSTS = new Set(["youtube.com", "www.youtube.com", "m.youtube.com", "youtu.be"]);

       /**
        * D-15 contract: returns ParsedUrl for any YouTube URL we recognize as a
        * standalone watchable item (video / short / embed / youtu.be link), null
        * otherwise. Channel and playlist URLs return null because the Phase 03.0
        * data model only ingests events of kind=youtube_video.
        */
       export function youtubeParseUrl(input: string): ParsedUrl | null {
         let url: URL;
         try {
           url = new URL(input.trim());
         } catch {
           return null;
         }
         const host = url.hostname.toLowerCase();
         if (!YT_HOSTS.has(host)) return null; // Pitfall 2 mitigation: host check FIRST

         // youtu.be/<videoId>
         if (host === "youtu.be") {
           const videoId = url.pathname.replace(/^\//, "").split("/")[0];
           if (!videoId) return null;
           return makeParsed(videoId, input);
         }

         // youtube.com /watch?v=<videoId>
         if (url.pathname === "/watch") {
           const v = url.searchParams.get("v");
           if (!v) return null;
           return makeParsed(v, input);
         }

         // youtube.com /shorts/<videoId>
         const shortsMatch = url.pathname.match(/^\/shorts\/([\w-]+)/);
         if (shortsMatch) {
           return makeParsed(shortsMatch[1], input);
         }

         // youtube.com /embed/<videoId>
         const embedMatch = url.pathname.match(/^\/embed\/([\w-]+)/);
         if (embedMatch) {
           return makeParsed(embedMatch[1], input);
         }

         // /channel/, /@handle, /c/, /user/ — these are channels, not videos. Return null.
         return null;
       }

       function makeParsed(videoId: string, originalInput: string): ParsedUrl {
         const canonical = `https://www.youtube.com/watch?v=${videoId}`;
         return {
           kind: "youtube_video",
           externalId: videoId,
           metadata: { canonicalUrl: canonical, originalUrl: originalInput },
         };
       }

       // Cross-source URL router — Phase 03.0.1 D-15.
       // Iterates allAdapters in registration order; first non-null parseUrl
       // wins. Returns RoutedUrl with `kind: "unsupported"` if no adapter
       // matches. The friendly Reddit deferral message is NOT added here —

       export type RoutedUrl = ParsedUrl | { kind: "unsupported" };

       export function parseAnyUrl(input: string): RoutedUrl {
         for (const adapter of allAdapters) {
           const parsed = adapter.parseUrl(input);
           if (parsed !== null) return parsed;
         }
         return { kind: "unsupported" };
       }
