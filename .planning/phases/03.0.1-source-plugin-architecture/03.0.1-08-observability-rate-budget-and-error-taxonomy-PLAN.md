// Cron gets 80%; user gets 20%. Per-day reset via the youtube.quota_reset cron handler.

export async function chargedFetch(url: URL, picked: PickedKey, units: number, ctx: AdapterContext): Promise<Response> {
  const reservoir = ctx.origin === "cron" ? cronReservoir : userReservoir;
  try {
    await reservoir.consume(picked.apiKeyId, units);
  } catch (rejRes: any) {
    throw new AdapterError("YouTube reservoir exhausted for origin pool", {
      category: "rate-limited",
      retryAfterMs: rejRes?.msBeforeNext ?? 60_000,
      context: { apiKeyId: picked.apiKeyId, origin: ctx.origin, units },
    });
  }
  // ... existing fetchWithTimeout + incrementUsage code ...
  // Map status codes via AdapterError on non-2xx:
  if (resp.status === 403) {
    const reason = await extractQuotaReason(resp); // existing helper or inline parse
    if (reason === "quotaExceeded") {
      throw new AdapterError("YouTube daily quota exceeded", {
        category: "rate-limited",
        retryAfterMs: msUntilMidnightPacific(),
        context: { apiKeyId: picked.apiKeyId },
      });
    }
    throw new AdapterError("YouTube auth failed (operator-issue)", {
      category: "operator-issue",
      context: { apiKeyId: picked.apiKeyId, reason },
    });
  }
  if (resp.status === 404) throw new AdapterError("YouTube resource not found", { category: "not-found" });
  if (resp.status >= 500) throw new AdapterError("YouTube transient error", { category: "transient" });
  // For other non-2xx, fall through to caller (existing behavior).
  return resp;
}

       // YouTube observability — Phase 03.0.1 D-08.

       // Implements adapter.observability surface used by /admin/quota and by
       // /sources/[id] connection-status badges (Phase 6+).
       import { eq, gte, and, desc, inArray } from "drizzle-orm";
       import { db } from "$lib/server/db/client.js";
       import { youtubeServiceQuotaUsage } from "$lib/server/db/schema/index.js";
       import { auditLog } from "$lib/server/db/schema/index.js";

         AdapterObservability,
         ObservabilityDailyStats,
         ObservabilityAuditEntry,
       } from "$lib/sources/adapter.js";
       import { todayPacific } from "./quota.js";

       const DAILY_LIMIT_PER_KEY = 10_000; // YouTube Data API v3 default
       const THROTTLE_EIGHTY = 0.80;
       const THROTTLE_NINETYFIVE = 0.95;

       async function getDailyStats(date: Date): Promise<ObservabilityDailyStats> {
         const datePT = todayPacific(date); // existing helper in quota.ts; if signature differs, adapt
         const rows = await db
           .select()
           .from(youtubeServiceQuotaUsage)
           // eslint-disable-next-line tenant-scope/no-unfiltered-tenant-query -- public external data, no tenant scope
           .where(eq(youtubeServiceQuotaUsage.datePacific, datePT));

         const numKeys = (env.SERVICE_YOUTUBE_API_KEYS ?? "").split(",").filter(Boolean).length || 1;
         const dailyLimit = DAILY_LIMIT_PER_KEY * numKeys;
         const unitsUsed = rows.reduce((sum, r) => sum + (r.estimatedUnits ?? 0), 0);
         const pctOfDaily = dailyLimit > 0 ? unitsUsed / dailyLimit : 0;
         const throttleState: "ok" | "eighty" | "ninetyfive" =
           pctOfDaily >= THROTTLE_NINETYFIVE ? "ninetyfive" :
           pctOfDaily >= THROTTLE_EIGHTY    ? "eighty"      : "ok";

         return {
           unitsUsed,
           dailyLimit,
           pctOfDaily,
           throttleState,
           keys: rows.map((r) => ({ apiKeyId: r.apiKeyId, unitsUsed: r.estimatedUnits ?? 0 })),

       async function getRecentAudit(limit: number): Promise<ObservabilityAuditEntry[]> {
         const rows = await db

           .orderBy(desc(auditLog.createdAt))

           action: r.action,
           occurredAt: r.occurredAt,
           metadata: (r.metadata as Record<string, unknown>) ?? {},
