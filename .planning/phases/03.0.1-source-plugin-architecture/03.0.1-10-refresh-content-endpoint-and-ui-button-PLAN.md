Endpoint shape (Hono, NOT SvelteKit):
```typescript
// Inside sources.ts (existing Hono sub-router; ADD this route to it):
sourcesRoutes.post(
  "/sources/:id/refresh-content",
  async (c) => {
    const userId = c.var.userId; // populated by tenantScope middleware (already mounted at /api/* router level)
    const sourceId = c.req.param("id");

    // findSource throws NotFoundError on cross-tenant (Phase 2.1 service contract).
    const source = await getSourceById(userId, sourceId);
    // service throws NotFoundError → mapErr in _shared.ts maps to 404 with body { error: "not_found" }
    // (NOT "forbidden" — preserves AGENTS.md invariant 2)

    if (source.deletedAt !== null) {
      throw new NotFoundError(); // soft-deleted → 404
    }

    let adapter;
    try {
      adapter = getAdapter(source.kind);
    } catch (e) {
      throw new AppError("Source kind not yet supported", "kind_not_yet_functional", 422);
    }

    const result = await adapter.backfillSource(
      { id: source.id, userId: source.userId, metadata: source.metadata as Record<string, unknown> },
      { userId, origin: "user" },
    );

    await writeAudit({
      userId,
      action: "source.refresh_content_requested",
      ipAddress: c.var.clientIp ?? null,
      userAgent: c.req.header("user-agent") ?? null,
      metadata: {
        source_id: source.id,
        kind: source.kind,
        queue: result.queue,
        job_id: result.jobId,
      },
    });

    return c.json({ enqueued: true, queue: result.queue, jobId: result.jobId }, 202);
  }
);
```

Backfill handler (NEW file src/lib/sources/youtube/server/handlers/backfill-user.ts):
```typescript
import type { PollableSource } from "$lib/sources/adapter.js";
import { logger } from "$lib/server/logger.js";
import { youtubeChannelAdapter as adapter } from "../adapter.js";
import { db } from "$lib/server/db/client.js";
import { dataSources, events } from "$lib/server/db/schema/index.js";
import { eq, and, isNull } from "drizzle-orm";

export async function handleBackfillUser(job: { data: { sourceId: string; userId: string; origin?: "user" | "cron" } }): Promise<void> {
  const { sourceId, userId } = job.data;
  // eslint-disable-next-line tenant-scope/no-unfiltered-tenant-query -- worker-internal lookup; userId in WHERE
  const sourceRow = await db.query.dataSources.findFirst({
    where: and(eq(dataSources.id, sourceId), eq(dataSources.userId, userId), isNull(dataSources.deletedAt)),
  });
  if (sourceRow == null) {
    logger.warn({ sourceId, userId }, "youtube.backfill.user job for missing/deleted source — skipping");
    return;
  }
  const since = sourceRow.lastPolledAt ?? new Date(Date.now() - 30 * 24 * 3600 * 1000); // 30d default if unknown
  const rawEvents = await adapter.pollContent(
    { id: sourceRow.id, userId: sourceRow.userId, metadata: (sourceRow.metadata as Record<string, unknown>) ?? {} },
    since,
    { userId, origin: "user" },
  );
  // Insert via the shared events service (not directly — events.ts owns INSERT discipline + dedup).
  // Use createEvent(userId, ...) for each rawEvent. Existing services/events.ts.createEvent
  // already handles author_is_me derivation and dedup of duplicate_event 23505.
  // ... loop ...
  logger.info({ sourceId, userId, count: rawEvents.length }, "youtube.backfill.user: events ingested");
}
```
