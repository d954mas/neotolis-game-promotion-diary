YouTube toCardProps mapper (RESEARCH.md Example 3 — verbatim contents to land in src/lib/sources/youtube/ui/card-props.ts):
```typescript
import type { CardProps } from "$lib/sources/card-props.js";

interface YoutubeEventLite {
  id: string;
  externalId: string | null;
  title: string;
  authorIsMe: boolean;
  stats?: { viewCount: number; likeCount: number; commentCount: number } | null;
}

export function toCardProps(event: YoutubeEventLite): CardProps {
  return {
    thumbnail: event.externalId ? `https://img.youtube.com/vi/${event.externalId}/mqdefault.jpg` : null,
    title: event.title,
    subtitle: event.authorIsMe ? "My video" : null,
    badge: event.authorIsMe ? "Mine" : null,
    metrics: event.stats ? [
      { label: "Views",    value: formatStat(event.stats.viewCount) },
      { label: "Likes",    value: formatStat(event.stats.likeCount) },
      { label: "Comments", value: formatStat(event.stats.commentCount) },
    ] : [],
    href: `/events/${event.id}`,
  };
}

function formatStat(n: number): string {
  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`;
  if (n >= 1_000)     return `${(n / 1_000).toFixed(1)}K`;
  return String(n);
}
```

Registry-UI wiring (replaces Plan 01 skeleton — src/lib/sources/registry-ui.ts):
```typescript
import type { SourceKind } from "./adapter.js";
import * as youtubeUiServer from "./youtube/ui/server.js";

export interface AdapterUiServer {
  toCardProps: (event: unknown) => import("./card-props.js").CardProps;
}

const uiRegistry = new Map<SourceKind, AdapterUiServer>([
  ["youtube_channel", youtubeUiServer],
]);

export function getAdapterUI(kind: SourceKind): AdapterUiServer {
  const ui = uiRegistry.get(kind);
  if (ui === undefined) throw new Error(`No UI registered for kind=${kind as string}`);
  return ui;
}
```

NOTE on event.kind vs source.kind: events have `kind: "youtube_video"` (event kind). The registry is keyed by SourceKind (`youtube_channel`). The mapping `eventKind → sourceKind` (e.g. `"youtube_video"` → `"youtube_channel"`) lives in a small helper:
```typescript
// src/lib/sources/event-to-source-kind.ts (NEW small helper)
import type { EventKind, SourceKind } from "./adapter.js";

export function eventKindToSourceKind(eventKind: EventKind): SourceKind | null {
  switch (eventKind) {
    case "youtube_video": return "youtube_channel";
    case "reddit_post":   return "reddit_account";
    case "twitter_post":  return "twitter_account";
    case "telegram_post": return "telegram_channel";
    case "discord_drop":  return "discord_server";
    // free-form / non-pollable kinds have no source kind:
    case "post":          return null;
    case "conference":    return null;
    case "talk":          return null;
    case "press":         return null;
    case "other":         return null;
  }
}
```

The /feed page passes `event.kind` to a helper that resolves the SourceKind via this mapping THEN calls getAdapterUI(sourceKind). Free-form kinds (with no source kind) fall back to a generic CardProps mapper inline in EventCard.svelte (or in cross-source default).
