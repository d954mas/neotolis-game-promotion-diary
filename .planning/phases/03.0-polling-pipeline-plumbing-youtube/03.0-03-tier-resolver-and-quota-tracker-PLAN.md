    ```typescript
    // Phase 3.0 Plan 03 — Pitfall 7 mitigation: single source of truth for tier resolution.
    // Every place that decides "what tier is this event?" imports from THIS module.
    // Inlining the boundary literals (e.g. `ageDays < 1`) elsewhere is a P0 review block.

    export type Tier = 'active' | 'cold' | 'frozen' | 'unavailable';

    /** D-05 boundary — Active tier is age < 24 hours. */
    export const TIER_BOUNDARY_ACTIVE_MS = 86_400_000;

    /** D-05 boundary — Cold tier is age < 28 days. Beyond = Frozen. */
    export const TIER_BOUNDARY_COLD_MS = 28 * 86_400_000;

    /** D-12 — last_poll_status values that override age-based tier with 'unavailable'. */
    export const UNAVAILABLE_POLL_STATUSES: readonly string[] = [
      'not_found',
      'private',
      'auth_error',
    ];

    export function resolveTier(
      occurredAt: Date,
      lastPollStatus: string | null,
      now: Date = new Date(),
    ): Tier {
      if (lastPollStatus !== null && UNAVAILABLE_POLL_STATUSES.includes(lastPollStatus)) {
        return 'unavailable';
      }
      const ageMs = now.getTime() - occurredAt.getTime();
      if (ageMs < TIER_BOUNDARY_ACTIVE_MS) return 'active';
      if (ageMs < TIER_BOUNDARY_COLD_MS) return 'cold';
      return 'frozen';
    }
    ```

    ```typescript
    import { createHash } from 'node:crypto';
    import { sql } from 'drizzle-orm';
    import { db } from '../db/client.js';
    import { youtubeServiceQuotaUsage } from '../db/schema/youtube-service-quota-usage.js';
    import { auditLog } from '../db/schema/audit-log.js';
    import { env } from '../config/env.js';
    import type { PgTransaction } from 'drizzle-orm/pg-core';

    export type ThrottleState = 'ok' | 'eighty' | 'ninetyfive';

    export interface PickedKey {
      apiKey: string;
      apiKeyId: string;
    }

    /** D-NEW idempotency — module-level audit-emission guard. Resets on container restart;
     *  cron `youtube.quota_reset` clears at midnight Pacific. */
    const auditedTransitions = new Map<string, Set<'eighty' | 'ninetyfive'>>();

    /** Round-robin index into env.SERVICE_YOUTUBE_API_KEYS. Module-level so cycling
     *  persists across calls within one worker process. */
    let roundRobinIdx = 0;

    /** D-13 — sha-8 of the API key string. Stored in youtube_service_quota_usage as identifier;
     *  also surfaced in /admin/quota table column. Stable across boots; collision-free at indie scale. */
    export function hashApiKeyId(apiKey: string): string {
      return createHash('sha256').update(apiKey).digest('hex').slice(0, 8);
    }

    /** D-13 — round-robin pick across SERVICE_YOUTUBE_API_KEYS. Returns null if env empty
     *  (auto-import + scheduled polling disabled — caller decides what to do). */
    export function pickKeyForJob(): PickedKey | null {
      const keys = env.SERVICE_YOUTUBE_API_KEYS;
      if (keys.length === 0) return null;
      const apiKey = keys[roundRobinIdx % keys.length];
      roundRobinIdx = (roundRobinIdx + 1) % keys.length;
      return { apiKey, apiKeyId: hashApiKeyId(apiKey) };
    }

    /** Pitfall D — YouTube quota resets midnight Pacific. Returns 'YYYY-MM-DD' in PT. */
    export function todayPacific(now: Date = new Date()): string {
      // Use Intl with America/Los_Angeles. Output sv-SE locale for ISO 'YYYY-MM-DD' shape.
      return new Intl.DateTimeFormat('sv-SE', {
        timeZone: 'America/Los_Angeles',
        year: 'numeric', month: '2-digit', day: '2-digit',
      }).format(now);
    }

    /** D-13 — UPSERT (date_pacific, api_key_id) += units. Idempotent at the row level. */
    export async function incrementUsage(args: {
      apiKeyId: string;
      units: number;
      tx?: PgTransaction<any, any, any>;
    }): Promise<void> {
      const datePacific = todayPacific();
      const dbCtx = args.tx ?? db;
      await dbCtx.insert(youtubeServiceQuotaUsage).values({
        datePacific,
        apiKeyId: args.apiKeyId,
        estimatedUnits: args.units,
      }).onConflictDoUpdate({
        target: [youtubeServiceQuotaUsage.datePacific, youtubeServiceQuotaUsage.apiKeyId],
        set: {
          estimatedUnits: sql`${youtubeServiceQuotaUsage.estimatedUnits} + ${args.units}`,
          updatedAt: new Date(),
        },
      });
    }

    /** D-13 — at-enqueue scheduler check. >=9500 = ninetyfive (pause everything but refresh-now);
     *  >=8000 = eighty (pause Cold + auto-import); else ok. */
    export async function getThrottleState(now: Date = new Date()): Promise<ThrottleState> {
      const datePacific = todayPacific(now);
      const rows = await db.select({ units: youtubeServiceQuotaUsage.estimatedUnits })
        .from(youtubeServiceQuotaUsage)
        .where(sql`${youtubeServiceQuotaUsage.datePacific} = ${datePacific}`);
      const max = rows.length === 0 ? 0 : Math.max(...rows.map((r) => r.units));
      if (max >= 9500) return 'ninetyfive';
      if (max >= 8000) return 'eighty';
      return 'ok';
    }

    /** D-13 — write audit `quota.service_throttled` ONCE per (date_pacific, state).
     *  Module-level Map gates re-emission; cron quota_reset clears at midnight PT
     *  via resetThrottleState (called by quota-reset worker). */
    export async function markThrottleTransition(args: {
      state: 'eighty' | 'ninetyfive';
      apiKeyId: string;
      estimatedUnits: number;
    }): Promise<void> {
      const datePacific = todayPacific();
      const seen = auditedTransitions.get(datePacific) ?? new Set<'eighty' | 'ninetyfive'>();
      if (seen.has(args.state)) return; // idempotent within container lifetime

      // Defense in depth — also check audit_log for prior emit (handles container restarts).
      const priorRows = await db.select({ id: auditLog.id }).from(auditLog).where(sql`
        ${auditLog.action} = 'quota.service_throttled'
        AND ${auditLog.metadata}->>'date_pacific' = ${datePacific}
        AND ${auditLog.metadata}->>'state' = ${args.state}
        LIMIT 1
      `);
      if (priorRows.length > 0) {
        seen.add(args.state);
        auditedTransitions.set(datePacific, seen);
        return;
      }

      // Service-level audit row — no per-tenant userId (operator's quota is global).
      // Per RESEARCH §"Pitfall 19" + Open Question 4: write under a synthetic system user
      // OR with userId = NULL if schema permits. Phase 02.2 audit_log requires user_id NOT NULL —
      // route through env.OPERATOR_USER_ID if available; otherwise pick the row by
      // resolving operator email → user_id via sessions/user table at boot.
      // PRACTICAL: Plan 03 stores under a recognised operator user_id resolved by a helper
      // (see code below). If allowlist is empty, we still record the audit row with the
      // first non-empty session user found OR drop the row with a logger.warn (parity gate
      // remains satisfied — empty allowlist self-host installs simply don't see /admin/quota
      // and the audit row absence is fine).
      const operatorId = await resolveOperatorUserId();
      if (operatorId === null) {
        const { logger } = await import('../logger.js');
        logger.warn(
          { datePacific, state: args.state },
          'quota.service_throttled threshold crossed but no operator user_id resolvable',
        );
        seen.add(args.state);
        auditedTransitions.set(datePacific, seen);
        return;
      }

      const { writeAudit } = await import('../audit.js');
      await writeAudit({
        userId: operatorId,
        action: 'quota.service_throttled',
        metadata: {
          date_pacific: datePacific,
          state: args.state,
          api_key_id: args.apiKeyId,
          estimated_units: args.estimatedUnits,
        },
      });

      seen.add(args.state);
      auditedTransitions.set(datePacific, seen);
    }

    /** Called by youtube.quota_reset cron handler (Plan 09) at midnight Pacific. */
    export function resetThrottleState(): void {
      auditedTransitions.clear();
      roundRobinIdx = 0;
    }

    /** Resolve operator's user_id by email. Picks first allowlist entry as canonical
     *  operator. Cached at module scope (one-time lookup; container restart re-resolves). */
    let cachedOperatorId: string | null | undefined; // undefined = not yet resolved
    async function resolveOperatorUserId(): Promise<string | null> {
      if (cachedOperatorId !== undefined) return cachedOperatorId;
      const allowlist = [...env.ADMIN_EMAIL_ALLOWLIST];
      if (allowlist.length === 0) {
        cachedOperatorId = null;
        return null;
      }
      const { user } = await import('../db/schema/auth.js');
      const rows = await db.select({ id: user.id }).from(user).where(sql`
        lower(${user.email}) = ${allowlist[0]} LIMIT 1
      `);
      cachedOperatorId = rows[0]?.id ?? null;
      return cachedOperatorId;
    }
    ```
