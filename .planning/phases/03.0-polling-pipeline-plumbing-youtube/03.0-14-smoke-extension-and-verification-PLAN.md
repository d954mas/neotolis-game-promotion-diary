    // Returns canned videos.list / playlistItems.list / channels.list responses.
    // Smoke shell sets YOUTUBE_API_BASE_URL to point here.

      kind: 'youtube#videoListResponse',
      items: Array.from({ length: 50 }, (_, i) => ({
        id: `mock-video-${i}`,
        snippet: {
          publishedAt: new Date(Date.now() - i * 86_400_000).toISOString(),
          title: `Mock Video ${i}`,
          channelId: 'UCmockchannel0123456789ab',
        },
        statistics: {
          viewCount: String(1000 + i * 100),
          likeCount: String(50 + i * 5),
          commentCount: String(10 + i),
        },
        contentDetails: {
          duration: i % 5 === 0 ? 'PT45S' : 'PT4M13S', // every 5th = Short
        },
      })),
    };

    const PLAYLIST_ITEMS_RESPONSE = {
      kind: 'youtube#playlistItemListResponse',
      items: Array.from({ length: 10 }, (_, i) => ({
        snippet: {
          publishedAt: new Date(Date.now() - i * 86_400_000).toISOString(),
          title: `Mock Upload ${i}`,
          channelId: 'UCmockchannel0123456789ab',
          resourceId: { kind: 'youtube#video', videoId: `mock-video-${i}` },
        },
      })),
    };

    const CHANNELS_RESPONSE = {
      kind: 'youtube#channelListResponse',
      items: [{
        id: 'UCmockchannel0123456789ab',
        snippet: { title: 'Mock Channel' },
        contentDetails: { relatedPlaylists: { uploads: 'UUmockchannel0123456789ab' } },
      }],
    };

    const server = createServer((req, res) => {
      const url = new URL(req.url, `http://localhost:${PORT}`);
      res.setHeader('Content-Type', 'application/json');
      if (url.pathname.endsWith('/videos')) {
        res.end(JSON.stringify(VIDEOS_RESPONSE));
      } else if (url.pathname.endsWith('/playlistItems')) {
        res.end(JSON.stringify(PLAYLIST_ITEMS_RESPONSE));
      } else if (url.pathname.endsWith('/channels')) {
        res.end(JSON.stringify(CHANNELS_RESPONSE));
      } else {
        res.statusCode = 404;
        res.end(JSON.stringify({ error: { code: 404, message: 'mock not implemented' } }));
      }
    });

    server.listen(PORT, () => {
      const addr = server.address();
      const boundPort = typeof addr === 'object' && addr ? addr.port : PORT;
      console.log(`youtube-mock listening on :${boundPort}`);
    });
    ```

    Create `tests/smoke/lib/youtube-mock.sh` (shell wrapper):

    #!/usr/bin/env bash
    # Phase 3.0 Plan 14 — wraps youtube-mock.mjs lifecycle for smoke gate.
    # Usage:
    #   source tests/smoke/lib/youtube-mock.sh
    #   start_youtube_mock     # exports YOUTUBE_MOCK_PORT + YOUTUBE_MOCK_PID
    #   ...assertions...
    #   stop_youtube_mock      # SIGTERM the mock; wait for exit

    YOUTUBE_MOCK_PID=""
    YOUTUBE_MOCK_PORT=""

    start_youtube_mock() {
      # Pick a random free port via Node (cross-platform; works on macOS + Linux smoke runners).
      YOUTUBE_MOCK_PORT=$(node -e 'const s = require("net").createServer(); s.listen(0, () => { console.log(s.address().port); s.close(); });')
      export YOUTUBE_MOCK_PORT
      node tests/smoke/lib/youtube-mock.mjs &
      YOUTUBE_MOCK_PID=$!
      export YOUTUBE_MOCK_PID
      sleep 1
      echo "youtube-mock started on :$YOUTUBE_MOCK_PORT (pid $YOUTUBE_MOCK_PID)"
    }

    stop_youtube_mock() {
      if [[ -n "$YOUTUBE_MOCK_PID" ]]; then
        kill -TERM "$YOUTUBE_MOCK_PID" 2>/dev/null || true
        wait "$YOUTUBE_MOCK_PID" 2>/dev/null || true
        echo "youtube-mock stopped"
      fi
    }

    # Auto-cleanup on EXIT.
    trap stop_youtube_mock EXIT
    ```
