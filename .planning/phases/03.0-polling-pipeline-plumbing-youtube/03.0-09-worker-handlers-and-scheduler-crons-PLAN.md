    ```typescript
    import { db } from '../../lib/server/db/client.js';
    import { events } from '../../lib/server/db/schema/events.js';
    import { youtubeChannelAdapter } from '../../lib/server/integrations/youtube-channel-adapter.js';
    import { writeSnapshot } from '../../lib/server/services/youtube-snapshot-writer.js';
    import { pickKeyForJob, markThrottleTransition } from '../../lib/server/services/youtube-quota-tracker.js';
    import { logger } from '../../lib/server/logger.js';

    /** Phase 3.0 Plan 09 — Active-tier poll handler.
     *  Two-phase tx pattern (Pitfall 5):
     *    A. Short SELECT (no FOR UPDATE) — load events
     *    B. HTTP via adapter (OUTSIDE any tx)
     *    C. writeSnapshot per event in single short write tx
     */
    export async function handlePollActive(job: { id: string; data: { eventIds: string[] } }): Promise<void> {
      const { eventIds } = job.data;
      if (!eventIds || eventIds.length === 0) return;

      // Phase A — load.
      const rows = await db.select({
        id: events.id,
        userId: events.userId,
        externalId: events.externalId,
      }).from(events).where(and(
        inArray(events.id, eventIds),
        isNull(events.deletedAt),
      ));
      const pollable = rows.filter((r) => r.externalId !== null) as { id: string; userId: string; externalId: string }[];
      if (pollable.length === 0) return;

      // Phase B — HTTP (outside tx).
      const picked = pickKeyForJob();
      if (!picked) {
        logger.warn({ jobId: job.id }, 'poll-active: SERVICE_YOUTUBE_API_KEYS empty; degrading to auth_error');
      }
      const snapshots = await youtubeChannelAdapter.pollStats(pollable, null);

      // Phase C — write per result.
      for (let i = 0; i < pollable.length; i++) {
        const ev = pollable[i];
        const snap = snapshots[i];
        await writeSnapshot({
          videoId: ev.externalId,
          eventId: ev.id,
          userId: ev.userId,
          metrics: snap.status === 'ok' && snap.metrics ? {
            view_count: snap.metrics.view_count ?? 0,
            like_count: snap.metrics.like_count ?? 0,
            comment_count: snap.metrics.comment_count ?? 0,
          } : null,
          apiKeyId: picked?.apiKeyId ?? 'no-key',
          unitsUsed: picked && snap.status !== 'auth_error' ? 1 : 0,
          status: snap.status,
        });

        if (snap.status === 'rate_limited' && picked) {
          // Mark threshold transition (idempotent per day per state).
          await markThrottleTransition({
            state: 'ninetyfive',
            apiKeyId: picked.apiKeyId,
            estimatedUnits: 9500,
          });
        }
      }
    }
    ```

    ```typescript
      const throttle = await getThrottleState();
      if (throttle === 'ninetyfive') {
        logger.info({ throttle }, 'scheduler skip Active — quota at 95%');
        return;
      }
      // 80% pauses Cold; Active continues.

      // Find events in Active range AND of pollable kinds AND not Unavailable.
      const now = new Date();
      const cutoff = new Date(now.getTime() - TIER_BOUNDARY_ACTIVE_MS);
      const rows = await db.select({
        id: events.id,
        userId: events.userId,
        occurredAt: events.occurredAt,
        lastPollStatus: events.lastPollStatus,
        sourceId: events.sourceId,
      }).from(events).where(and(
        eq(events.kind, 'youtube_video'),
        gt(events.occurredAt, cutoff),
        isNotNull(events.externalId),
        isNull(events.deletedAt),
      ));

      // Filter via resolveTier (defense in depth — SQL window may include Unavailable rows).
      const eligible = rows.filter((r) => resolveTier(r.occurredAt, r.lastPollStatus, now) === 'active');

      // Auto-import path needs data_sources.auto_import = true; manual paste (sourceId=NULL) always polls.
      // ... join data_sources for sourceId-bearing events; filter accordingly.

      const boss = await getBoss();
      // Group into batches of 50 (videos.list limit).
      for (let i = 0; i < eligible.length; i += 50) {
        const batch = eligible.slice(i, i + 50).map((e) => e.id);
        await boss.send(QUEUES.POLL_ACTIVE, { eventIds: batch }, {
          singletonKey: `active-${batch[0]}-${batch.length}`,
        });
      }
    }

    export async function enqueueColdPolls(): Promise<void> {
      const throttle = await getThrottleState();
      if (throttle !== 'ok') {
        logger.info({ throttle }, 'scheduler skip Cold — quota throttled');
        return;
      }
    ```
