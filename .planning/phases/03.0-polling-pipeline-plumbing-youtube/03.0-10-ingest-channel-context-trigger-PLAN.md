    import { db } from '../db/client.js';
    import { youtubeChannelMetadataCache } from '../db/schema/youtube-channel-metadata-cache.js';
    import { QUEUES } from '../queues.js';
    import { getBoss } from '../queue-client.js';
    import { logger } from '../logger.js';

    // Inside the existing youtube_video paste handler, AFTER event creation, BEFORE returning:
    // (Pseudo-code — adapt to existing function shape.)
    if (kind === 'youtube_video' && oembed?.authorUrl) {
      const channelId = extractChannelIdFromAuthorUrl(oembed.authorUrl);
      if (channelId) {
        const cached = await db.select({ channelId: youtubeChannelMetadataCache.channelId })
          .from(youtubeChannelMetadataCache)
          .where(eq(youtubeChannelMetadataCache.channelId, channelId))
          .limit(1);
        if (cached.length === 0) {
          // Cache miss — enqueue one-time backfill (idempotent via singletonKey).
          try {
            const boss = await getBoss();
            await boss.send(QUEUES.YOUTUBE_CHANNEL_CONTEXT_BACKFILL, {
              channelId,
              userId,
            }, {
              singletonKey: channelId, // dedupes concurrent pastes from same channel
              singletonHours: 24,
            });
          } catch (err) {
            // Fire-and-forget — backfill is a quality-of-Phase-4-charts enhancement, not load-bearing.
            logger.warn({ channelId, err }, 'channel-context-backfill enqueue failed; ignoring');
          }
        }
      }
    }

    /** D-14 — extract channelId from oEmbed author_url.
     *  YouTube author_url shapes:
     *    https://www.youtube.com/@handle   (no UC channel id; the backfill handler resolves via channels.list?forHandle=)
     *    https://www.youtube.com/channel/UCabc... (direct channelId)
     *    https://www.youtube.com/c/customname (legacy; resolved via search)
     *  Phase 3.0: only the direct /channel/UC... shape is used as singletonKey here;
     *  for /@handle and /c/ shapes, return the URL itself as the singletonKey (safe — same
     *  URL = same paste = same dedup) and let the backfill handler do the lookup.
     */
    function extractChannelIdFromAuthorUrl(authorUrl: string): string | null {
      const m = /\/channel\/(UC[A-Za-z0-9_-]{20,})/.exec(authorUrl);
      if (m) return m[1];
      // Fallback: use the full URL as the singletonKey identifier; backfill handler resolves.
      // Cache lookup will miss (channelId column has UC... format), so backfill always runs
      // for /@handle URLs — acceptable: 2 quota units one-time per channel.
      return authorUrl;
    }
