    ```typescript
    import { requestRefreshPoll } from '../../services/refresh-poll.js';
    import { AppError } from '../../services/errors.js';

    // Existing eventsRouter mounted somewhere; locate and add:
    eventsRouter.post('/:id/refresh-poll', async (c) => {
      try {
        const result = await requestRefreshPoll(c.var.userId, c.req.param('id'));
        return c.json(result);
      } catch (err) {
        // Set Retry-After header for 429 if metadata carries retryAfterSeconds.
        if (err instanceof AppError && err.code === 'too_many_refreshes') {
          const meta = err.metadata as { retryAfterSeconds?: number } | undefined;
          if (meta?.retryAfterSeconds != null) {
            c.header('Retry-After', String(meta.retryAfterSeconds));
          }
        }
        return mapErr(c, err);
      }
    });
    ```
