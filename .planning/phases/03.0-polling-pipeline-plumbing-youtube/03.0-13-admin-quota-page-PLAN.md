    import type { PageServerLoad } from './$types';

    export const load: PageServerLoad = async ({ fetch }) => {
      const resp = await fetch('/api/admin/quota');
      if (resp.status === 404) {
        throw error(404, 'Not Found');
      }
      if (!resp.ok) {
        throw error(500, 'admin quota load failed');
      }
      return await resp.json() as { today: string; keys: any[]; audit: any[] };
    };

    <script lang="ts">
      import * as m from '$lib/paraglide/messages';
      import EmptyState from './EmptyState.svelte';

      type ServiceAuditEntry = {
        id: string;
        createdAt: Date | string;
        action: string;
        metadata: Record<string, unknown>;
      };

      let { entries }: { entries: ServiceAuditEntry[] } = $props();

      function fmt(d: Date | string): string {
        const date = typeof d === 'string' ? new Date(d) : d;
        // Match Phase 2 audit row date format — locale-fixed for SSR consistency.
        return date.toISOString().slice(0, 19).replace('T', ' ');
      }

      function metadataSummary(meta: Record<string, unknown>): string {
        // Compact one-line summary — operator can drill into full audit log via /audit elsewhere.
        const keys = Object.keys(meta);
        if (keys.length === 0) return '';
        return keys.slice(0, 3).map((k) => `${k}=${JSON.stringify(meta[k])}`).join(' · ');
      }
    </script>

    {#if entries.length === 0}
      <EmptyState heading={m.admin_quota_audit_empty_heading()} body={m.admin_quota_audit_empty_body()} />
    {:else}
      <ul class="quota-audit-list">
        {#each entries as e (e.id)}
          <li class="quota-audit-row">
            <time datetime={typeof e.createdAt === 'string' ? e.createdAt : e.createdAt.toISOString()}>
              {fmt(e.createdAt)}
            </time>
            <code class="action">{e.action}</code>
            <span class="meta">{metadataSummary(e.metadata)}</span>
          </li>
        {/each}
      </ul>
    {/if}

    <style>
      .quota-audit-list { list-style: none; padding: 0; margin: 0; display: flex; flex-direction: column; }
