    import { db } from '../db/client.js';
    import { youtubeVideoSnapshots } from '../db/schema/youtube-video-snapshots.js';
    import { events } from '../db/schema/events.js';
    import { incrementUsage } from './youtube-quota-tracker.js';

    export type SnapshotStatus = 'ok' | 'not_found' | 'private' | 'auth_error' | 'rate_limited';

    /**
     * Phase 3.0 Plan 04 — idempotent snapshot writer (POLL-04 + Pitfall 5).
     *
     * Atomic across 3 ops in a single tx:
     *   1. INSERT row in youtube_video_snapshots with polled_at truncated to minute
     *      (UNIQUE (video_id, polled_at) ON CONFLICT DO NOTHING — within-the-minute retries no-op)
     *   2. UPDATE events.last_polled_at + last_poll_status (tenant-scoped via userId)
     *   3. UPSERT youtube_service_quota_usage += unitsUsed (per-key per-day counter)
     *
     * Caller is responsible for the HTTP call (which happens OUTSIDE this tx — Pitfall 5).
     * This service expects the metrics + status as already-resolved inputs.
     */
    export async function writeSnapshot(args: {
      videoId: string;
      eventId: string;
      userId: string;
      metrics: { view_count: number; like_count: number; comment_count: number } | null;
      apiKeyId: string;
      unitsUsed: number;
      status: SnapshotStatus;
    }): Promise<void> {
      await db.transaction(async (tx) => {
        if (args.status === 'ok' && args.metrics !== null) {
          await tx.insert(youtubeVideoSnapshots).values({
            videoId: args.videoId,
            polledAt: sql`date_trunc('minute', now())` as unknown as Date,
            viewCount: args.metrics.view_count,
            likeCount: args.metrics.like_count,
            commentCount: args.metrics.comment_count,
          }).onConflictDoNothing();
        }

        // ESLint tenant-scope: events table is tenant-owned — eq(events.userId, ...) required.
        await tx.update(events)
          .set({
            lastPolledAt: new Date(),
            lastPollStatus: args.status,
          })
          .where(and(eq(events.id, args.eventId), eq(events.userId, args.userId)));

        // Increment per-key counter inside the same tx — pass `tx` explicitly to incrementUsage.
        await incrementUsage({
          apiKeyId: args.apiKeyId,
          units: args.unitsUsed,
          tx,
        });
      });
    }
    ```

    import { eq, and, isNull } from 'drizzle-orm';
    import { db } from '../db/client.js';
    import { events } from '../db/schema/events.js';
    import { AppError, NotFoundError } from './errors.js';
    import { writeAudit } from '../audit.js';
    import { QUEUES } from '../queues.js';
    import { getBoss } from '../queue-client.js';
    // ^ if queue-client.ts exposes getBoss; else use the module-level boss singleton pattern
    //   established by Phase 1 worker.

    const COOLDOWN_MS = 5 * 60 * 1000;

    export interface RefreshPollResult {
      enqueued: true;
      queue: 'poll.user';
      eventId: string;
    }

    export async function requestRefreshPoll(
      userId: string,
      eventId: string,
    ): Promise<RefreshPollResult> {
      const rows = await db.select().from(events).where(and(
        eq(events.id, eventId),
        eq(events.userId, userId),
        isNull(events.deletedAt),
      ));
      const event = rows[0];
      if (!event) throw new NotFoundError();

      // D-10: only kind=youtube_video is pollable in Phase 3.0 (reddit_post is Phase 3.1).
      if (event.kind !== 'youtube_video') {
        throw new AppError(422, 'event_not_pollable', {});
      }
      if (!event.externalId) {
        throw new AppError(422, 'event_no_external_id', {});
      }

      // 5-min cooldown gate via events.metadata.last_user_refresh_at.
      const meta = (event.metadata ?? {}) as Record<string, unknown>;
      const lastRefreshIso = typeof meta.last_user_refresh_at === 'string'
        ? meta.last_user_refresh_at
        : null;
      const lastRefreshAt = lastRefreshIso ? new Date(lastRefreshIso) : null;
      const now = new Date();
      if (lastRefreshAt && now.getTime() - lastRefreshAt.getTime() < COOLDOWN_MS) {
        const msLeft = COOLDOWN_MS - (now.getTime() - lastRefreshAt.getTime());
        const minutesLeft = Math.ceil(msLeft / 60000);
        throw new AppError(429, 'too_many_refreshes', {
          minutesLeft,
          retryAfterSeconds: Math.ceil(msLeft / 1000),
        });
      }

      // Persist last_user_refresh_at BEFORE enqueueing so a crash mid-enqueue still
      // honors the cooldown on the next attempt (eager-write — Phase 02.2 quota pattern).
      await db.update(events).set({
        metadata: { ...meta, last_user_refresh_at: now.toISOString() },
      }).where(and(eq(events.id, eventId), eq(events.userId, userId)));

      // Enqueue with singletonKey scoped to per-minute window (prevents accidental flood).
      const boss = await getBoss();
      const minuteKey = now.toISOString().slice(0, 16); // 'YYYY-MM-DDTHH:MM'
      await boss.send(QUEUES.POLL_USER, { eventId, userId }, {
        singletonKey: `${eventId}-${minuteKey}`,
        priority: 10, // ahead of scheduled work
      });

      // Audit (tenant-scoped to the event owner).
      await writeAudit({
        userId,
        action: 'event.poll_refreshed',
        metadata: { event_id: eventId, kind: event.kind },
      });

      return { enqueued: true, queue: 'poll.user', eventId };
    }
    ```
