    <script lang="ts">
      import * as m from '$lib/paraglide/messages';
      import { invalidateAll } from '$app/navigation';

      type EventForRefresh = {
        id: string;
        metadata: Record<string, unknown> | null;
      };

      let { event }: { event: EventForRefresh } = $props();

      type State = 'idle' | 'pending' | 'cooldown' | 'done' | 'error';
      let state = $state<State>('idle');
      let cooldownMinutesLeft = $state(0);
      let cooldownTimer: ReturnType<typeof setInterval> | null = null;

      const COOLDOWN_MS = 5 * 60 * 1000;

      function computeCooldown(): number {
        const meta = (event.metadata ?? {}) as { last_user_refresh_at?: string };
        if (!meta.last_user_refresh_at) return 0;
        const last = new Date(meta.last_user_refresh_at).getTime();
        const elapsed = Date.now() - last;
        if (elapsed >= COOLDOWN_MS) return 0;
        return Math.ceil((COOLDOWN_MS - elapsed) / 60_000);
      }

      // Initial cooldown evaluation + 30s interval re-evaluation.
      $effect(() => {
        cooldownMinutesLeft = computeCooldown();
        if (cooldownMinutesLeft > 0) state = 'cooldown';
        cooldownTimer = setInterval(() => {
          cooldownMinutesLeft = computeCooldown();
          if (cooldownMinutesLeft === 0 && state === 'cooldown') state = 'idle';
        }, 30_000);
        return () => {
          if (cooldownTimer) clearInterval(cooldownTimer);
        };
      });

      async function handleClick() {
        if (state !== 'idle') return;
        state = 'pending';
        try {
          const resp = await fetch(`/api/events/${event.id}/refresh-poll`, { method: 'POST' });
          if (resp.status === 200 || resp.status === 202) {
            state = 'done';
            setTimeout(() => {
              state = 'idle';
              invalidateAll();
            }, 2000);
            return;
          }
          if (resp.status === 429) {
            const retryAfter = resp.headers.get('Retry-After');
            const seconds = retryAfter ? Number(retryAfter) : COOLDOWN_MS / 1000;
            cooldownMinutesLeft = Math.ceil(seconds / 60);
            state = 'cooldown';
            return;
          }
          state = 'error';
        } catch {
          state = 'error';
        }
      }

      const disabled = $derived(state === 'pending' || state === 'cooldown' || state === 'done');
      const tooltip = $derived(
        state === 'cooldown'
          ? m.polling_refresh_now_cooldown_tooltip({ minutesLeft: cooldownMinutesLeft })
          : ''
      );
    </script>

    <button
      type="button"
      class="refresh-now"
      class:refresh-now--pending={state === 'pending'}
      class:refresh-now--cooldown={state === 'cooldown'}
      class:refresh-now--done={state === 'done'}
      aria-label={m.polling_refresh_now_aria()}
      aria-busy={state === 'pending'}
      aria-describedby={state === 'cooldown' ? 'refresh-cooldown-tip' : undefined}
      {disabled}
      onclick={handleClick}
    >
      <svg class="refresh-now__icon" viewBox="0 0 24 24" width="16" height="16"
           stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round" fill="none"
           aria-hidden="true">
        <!-- Circular-arrow refresh glyph per UI-SPEC §"Inline-SVG icon additions" -->
        <path d="M3 12a9 9 0 0 1 15-6.7L21 8" />
        <path d="M21 3v5h-5" />
        <path d="M21 12a9 9 0 0 1-15 6.7L3 16" />
        <path d="M3 21v-5h5" />
      </svg>
      {#if state === 'cooldown'}
        <span id="refresh-cooldown-tip" class="sr-only">{tooltip}</span>
      {/if}
      {#if state === 'pending'}
        <span class="sr-only" role="status">{m.polling_refresh_now_pending()}</span>
      {/if}
    </button>
    {#if state === 'error'}
      <small role="alert" class="refresh-now-error">{m.polling_refresh_now_error()}</small>
    {/if}

      .refresh-now {
