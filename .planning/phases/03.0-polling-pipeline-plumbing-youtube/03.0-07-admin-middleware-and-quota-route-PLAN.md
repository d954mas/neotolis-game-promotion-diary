    ```typescript
    /**

     * Reads env.ADMIN_EMAIL_ALLOWLIST (Set<string> parsed at boot). Throws NotFoundError
     * (→ 404 via routes/_shared.ts mapErr) when caller email is not in the set.
     *
     * NEVER throws ForbiddenError on tenant-owned resources — AGENTS.md invariant 2 + AP-4.
     * Self-host parity by construction: empty allowlist Set ⇒ all /admin/* returns 404.
     *
     * Mount AFTER tenantScope so c.var.userEmail is populated; mount in app.ts as:

     */
    export const adminAllowlist: MiddlewareHandler<{
      Variables: { userId: string; sessionId: string; userEmail: string };
    }> = async (c, next) => {
      const email = c.var.userEmail.toLowerCase();

        throw new NotFoundError();

      return next();
    ```

    ```typescript
    import { sql, eq, inArray, desc } from 'drizzle-orm';
    import { db } from '../db/client.js';
    import { youtubeServiceQuotaUsage } from '../db/schema/youtube-service-quota-usage.js';
    import { auditLog } from '../db/schema/audit-log.js';
    import { todayPacific } from './youtube-quota-tracker.js';

    export interface QuotaKeyRow {
      apiKeyId: string;
      estimatedUnits: number;
      pctOfDaily: number;
      status: 'ok' | '80_throttle' | '95_throttle';
    }

    export interface ServiceAuditEntry {
      id: string;
      createdAt: Date;
      action: string;
      metadata: Record<string, unknown>;
    }

    /** Phase 3.0 Plan 07 — admin /admin/quota loader.
     *  Allowlist gate is the security property; this service does CROSS-TENANT reads
     *  intentionally (operator visibility into service-level state per CONTEXT D-16). */
    const SERVICE_LEVEL_AUDIT_ACTIONS = [
      'quota.service_throttled',
      'purge.completed',
      'auto_import.deferred',
      'poll.failed',
    ] as const;

    export async function loadAdminQuotaPage(): Promise<{
      today: string;
      keys: QuotaKeyRow[];
      audit: ServiceAuditEntry[];
    }> {
      const today = todayPacific();

      const usageRows = await db.select().from(youtubeServiceQuotaUsage)
        .where(eq(youtubeServiceQuotaUsage.datePacific, today));

      // eslint-disable-next-line tenant-scope/no-unfiltered-tenant-query -- /admin/quota is allowlist-gated; cross-tenant audit aggregation is the intended operator view (CONTEXT D-16 / DV-3 / Open Question 4).
      const auditRows = await db.select().from(auditLog)
        .where(inArray(auditLog.action, SERVICE_LEVEL_AUDIT_ACTIONS as unknown as string[]))
        .orderBy(desc(auditLog.createdAt))
        .limit(50);

      return {
        today,
        keys: usageRows.map(toQuotaKeyRow),
        audit: auditRows.map(toServiceAuditEntry),
      };
    }

    function toQuotaKeyRow(row: typeof youtubeServiceQuotaUsage.$inferSelect): QuotaKeyRow {
      const u = row.estimatedUnits;
      const pct = Math.round((u / 10_000) * 1000) / 10; // 1 decimal
      const status: QuotaKeyRow['status'] =
        u >= 9500 ? '95_throttle' :
        u >= 8000 ? '80_throttle' :
        'ok';
      return {
        apiKeyId: row.apiKeyId,
        estimatedUnits: u,
        pctOfDaily: pct,
        status,
      };
    }

    function toServiceAuditEntry(row: typeof auditLog.$inferSelect): ServiceAuditEntry {
      return {
        id: row.id,
        createdAt: row.createdAt,
        action: row.action,
        metadata: (row.metadata ?? {}) as Record<string, unknown>,
      };
    }
    ```
