    import { user, session } from '../db/schema/auth.js';

    import { writeAudit } from '../audit.js';
    import { env } from '../config/env.js';

    export interface PurgeOptions {
      ignoreRetention?: boolean;
    }

    export interface PurgeResult {
      purged: boolean;
      rowCounts?: {
        apiKeysSteam: number;
        events: number;

    /**
     * Phase 3.0 Plan 05 — closes Phase 02.2 PUTOFF + GDPR Art. 17 60-day-grace.
     *
     * Cascade chain in one tx, FK-respecting order:
     *   api_keys_steam → events → event_games → game_steam_listings →
     *   data_sources → games → session → user
     *
     * Idempotent: re-running on already-purged user returns {purged: true, rowCounts: all 0s}.
     * Audit row written OUTSIDE the tx (pool-deadlock-safe pattern from Phase 02.2 D-15).
     * Audit row scoped to the PURGED user_id (Open Question 4 resolution).
     */
    export async function purgeAccount(
      userId: string,
      opts: PurgeOptions = {},
    ): Promise<PurgeResult> {
      // Eligibility gate.
      if (!opts.ignoreRetention) {
        const rows = await db.select({ deletedAt: user.deletedAt }).from(user)
          .where(eq(user.id, userId));
        const deletedAt = rows[0]?.deletedAt;
        if (!deletedAt) return { purged: false };
        const ageMs = Date.now() - deletedAt.getTime();
        if (ageMs < env.RETENTION_DAYS * 86_400_000) return { purged: false };
      }

      // FK-respecting cascade DELETE in one tx.
      // Each step returns affected count via `.returning()` to populate row_counts.
      const counts = await db.transaction(async (tx) => {
        // 1. api_keys_steam (children of user, NOT cascaded by other tables)
        const ak = await tx.delete(apiKeysSteam).where(eq(apiKeysSteam.userId, userId)).returning({ id: apiKeysSteam.id });

        // 2. event_games (junction — children of events; delete first since FK to events.id)
        //    event_games.userId is denormalized (Plan 02.1-27); use it for tenant scope.
        const eg = await tx.delete(eventGames).where(eq(eventGames.userId, userId)).returning({ eventId: eventGames.eventId });

        // 3. events (child of user + data_sources; delete after event_games)
        const ev = await tx.delete(events).where(eq(events.userId, userId)).returning({ id: events.id });

        // 4. game_steam_listings (children of games)
        const gsl = await tx.delete(gameSteamListings).where(eq(gameSteamListings.userId, userId)).returning({ id: gameSteamListings.id });

        // 5. data_sources (children of user)
        const ds = await tx.delete(dataSources).where(eq(dataSources.userId, userId)).returning({ id: dataSources.id });

        // 6. games (children of user)
        const g = await tx.delete(games).where(eq(games.userId, userId)).returning({ id: games.id });

        // 7. session (Better Auth)
        const sess = await tx.delete(session).where(eq(session.userId, userId)).returning({ id: session.id });

        // 8. user (final)
        const u = await tx.delete(user).where(eq(user.id, userId)).returning({ id: user.id });

        return {
          apiKeysSteam: ak.length,
          events: ev.length,
          eventGames: eg.length,
          gameSteamListings: gsl.length,
          dataSources: ds.length,
          games: g.length,
          sessions: sess.length,
          user: u.length,
        };
      });

      // Audit OUTSIDE the tx — Phase 02.2 pool-deadlock-safe pattern.
      // Scoped to the purged user_id (Open Question 4).
      // NOTE: writeAudit may need a special path that allows orphan user_id since the user
      // row no longer exists post-tx. Check if audit_log has an FK to user — Phase 02.2
      // softDeleteAccount writes audit BEFORE soft-delete to avoid this problem.
      // For purge: audit_log.user_id has no FK constraint per Phase 1 schema (it's a text
      // column without REFERENCES); confirm in audit-log.ts schema before writing.
      // If FK exists, write audit BEFORE the tx and accept that a tx rollback leaves the
      // audit row as a "phantom" — acceptable per CONTEXT D-NEW Purge worker.
      await writeAudit({
        userId,
        action: 'purge.completed',
        metadata: {
          row_counts: counts,
          ignore_retention: opts.ignoreRetention === true,
          purged_at: new Date().toISOString(),
        },
      });

      return { purged: true, rowCounts: counts };
    }

    export async function listPurgeEligibleUsers(now: Date = new Date()): Promise<string[]> {
      const cutoff = new Date(now.getTime() - env.RETENTION_DAYS * 86_400_000);
      const rows = await db.select({ id: user.id }).from(user)
        .where(and(isNotNull(user.deletedAt), lt(user.deletedAt, cutoff)));
      return rows.map((r) => r.id);
    }
